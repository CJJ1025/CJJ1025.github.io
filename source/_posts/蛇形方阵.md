蛇形方阵是模拟类题目的经典入门题，核心在于按照「右 → 下 → 左 → 上」的顺序转圈填数。本文记录两种常见写法。

## 题目一：洛谷 [P5731 【深基5.习6】蛇形方阵](https://www.luogu.com.cn/problem/P5731)

### 方法一：4 个 while 循环

用四个 `while` 分别控制四个方向的填充，每个方向走到「越界」或「格子已有值」时转向。

```cpp
#include<iostream>
#include<vector> 
#include<cstdio>
using namespace std;
int main(){
	int n;
	cin>>n;
	vector<vector<int>> a(n+1,vector<int>(n+1,0)); 
	
	int y=0,x=0;
	//y行x列 
	for(int num=1;num<n*n;){ 
		//向右，x<n保证不超边界【0，n-1】，并保证填入的地方为空 
		while(x<n&&a[y][x]==0){
			a[y][x]=num;
			num++;
			x++;
		}
		x--;
		y++;
		
		while(y<n&&a[y][x]==0){//向下 
			a[y][x]=num;
			num++;
			y++;
		}
		y--;
		x--;
		
		while(x>=0&&a[y][x]==0){//向左 
			a[y][x]=num;
			num++;
			x--;
		}
		x++;
		y--;
		
		while(y>=0&&a[y][x]==0){//向上 
			a[y][x]=num;
			num++;
			y--;
		}
		y++;
		x++;
	}
	
	for(int i=0;i<n;i++){
		for(int j=0;j<n;j++){
			printf("%3d",a[i][j]);
		}
		cout<<endl;
	}
	
	
	return 0;
}
```

### 方法二：方向数组 + 循环 n×n 次

用 `dx/dy` 方向数组表示四个方向，每次先填数，再判断下一格是否越界或已被占用，若是则顺时针转向。相比方法一，代码结构更统一，也更容易扩展。

```cpp
#include<cstdio>
#include<vector>
using namespace std;
int main(){
	int n;
	scanf("%d",&n);
	int x=0,y=0;
	vector<vector<int>> a(n,vector<int>(n,0));
	
	//0右，1下，2左，3上 顺时针 
	int temp=0;
	int dx[4]={1,0,-1,0};
	int dy[4]={0,1,0,-1};
	
	for(int i=1;i<=n*n;i++){
		
		a[y][x]=i;
		
		int nx=x+dx[temp];
		int ny=y+dy[temp];
		//当到达边界时，或者下一步空位有值时，进到下一步 
		if(nx>=n||ny>=n||nx<0||ny<0||a[ny][nx]!=0){
			temp++;
			temp%=4;
			nx=x+dx[temp];
			ny=y+dy[temp];
		}
		x=nx;
		y=ny;
	}
	
	for(int i=0;i<n;i++){
		for(int j=0;j<n;j++){
			printf("%3d",a[i][j]);
		}
		printf("\n");
	}
	
	return 0;
}
```

### 两种写法对比

| 对比项 | 方法一：4 个 while | 方法二：方向数组 |
| --- | --- | --- |
| 代码结构 | 四段重复逻辑，靠回退坐标转向 | 统一循环，逻辑清晰 |
| 上手难度 | 直观，容易想到 | 需要理解方向数组 |
| 扩展性 | 加方向要再写一段 | 改数组即可，适合更复杂的模拟 |
